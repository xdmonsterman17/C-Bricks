class_name ShaderPalette
extends VBoxContainer
## The Shader screen's palette (phase 2): every node of `shader_nodes.json` that the open graph's kind
## can use, grouped, with a search box. Double-click or drag one onto the canvas to add it. Shown in
## place of the brick palette while a `.cbshader` is open (it has no Scene / FileSystem swap button).

signal add_requested(type: String)

var kind := "surface"
var _search: LineEdit = null
var _list: VBoxContainer = null
var _note: Label = null


func _init() -> void:
	custom_minimum_size.x = 0
	_note = Label.new()
	_note.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	_note.add_theme_color_override(&"font_color", Color(0.55, 0.8, 0.6))
	add_child(_note)
	_search = LineEdit.new()
	_search.placeholder_text = "Search all nodes"
	_search.clear_button_enabled = true
	_search.text_changed.connect(func(_t: String) -> void: refill())
	add_child(_search)
	var scroll := ScrollContainer.new()
	scroll.size_flags_vertical = SIZE_EXPAND_FILL
	scroll.horizontal_scroll_mode = ScrollContainer.SCROLL_MODE_DISABLED
	add_child(scroll)
	_list = VBoxContainer.new()
	_list.size_flags_horizontal = SIZE_EXPAND_FILL
	_list.add_theme_constant_override(&"separation", 4)
	scroll.add_child(_list)


func set_kind(k: String) -> void:
	kind = k
	refill()


func refill() -> void:
	_note.text = "Shader nodes (%s shader): the graph compiles to GLSL when you save it." % kind
	for child in _list.get_children():
		_list.remove_child(child)
		child.queue_free()
	var letter := "p" if kind == "post" else "s"
	var query := _search.text.strip_edges().to_lower()
	var tab := ShaderGraph.table()
	var groups := {}
	for type: String in tab:
		var def: Dictionary = tab[type]
		if str(def.kinds).find(letter) < 0:
			continue
		if query != "" and not ("%s %s %s" % [def.title, type, def.group]).to_lower().contains(query):
			continue
		if not groups.has(def.group):
			groups[def.group] = []
		groups[def.group].append(type)
	for group: String in ["Input", "Constant", "Math", "Vector", "Pattern", "Color", "Output"]:
		if not groups.has(group):
			continue
		var header := Label.new()
		header.text = group
		header.add_theme_color_override(&"font_color", ShaderCanvas.group_color(group))
		_list.add_child(header)
		for type: String in groups[group]:
			_list.add_child(_entry(type, tab[type], ShaderCanvas.group_color(group)))
	if groups.is_empty():
		var none := Label.new()
		none.text = "No nodes match \"%s\"." % _search.text
		_list.add_child(none)


func _entry(type: String, def: Dictionary, color: Color) -> Button:
	var button := Button.new()
	button.text = str(def.title)
	button.alignment = HORIZONTAL_ALIGNMENT_LEFT
	button.clip_text = true
	button.text_overrun_behavior = TextServer.OVERRUN_TRIM_ELLIPSIS
	var ins := ", ".join(def["in"].map(func(p: Array) -> String: return str(p[0])))
	var outs := ", ".join(def["out"].map(func(p: Array) -> String: return str(p[0])))
	button.tooltip_text = "%s\nin: %s\nout: %s\n\nDrag onto the canvas, or double-click, to add it." % [def.title, ins if ins != "" else "-", outs if outs != "" else "-"]
	for state: StringName in [&"normal", &"hover", &"pressed", &"focus"]:
		var style := StyleBoxFlat.new()
		style.bg_color = Color(0.14, 0.15, 0.17).lerp(color, 0.32 if state == &"hover" else 0.2)
		style.border_color = color
		style.border_width_left = 4
		style.set_corner_radius_all(4)
		style.content_margin_left = 10
		style.content_margin_right = 8
		style.content_margin_top = 3
		style.content_margin_bottom = 3
		button.add_theme_stylebox_override(state, style)
	button.mouse_default_cursor_shape = Control.CURSOR_DRAG
	button.set_drag_forwarding(func(_at: Vector2) -> Variant:
		var ghost := Label.new()
		ghost.text = str(def.title)
		ghost.modulate.a = 0.7
		button.set_drag_preview(ghost)
		return {ShaderCanvas.DRAG_KEY: type}, Callable(), Callable())
	button.gui_input.connect(func(event: InputEvent) -> void:
		var click := event as InputEventMouseButton
		if click != null and click.double_click and click.button_index == MOUSE_BUTTON_LEFT:
			add_requested.emit(type))
	return button
